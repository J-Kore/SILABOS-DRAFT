// ─── Insignia Orión — animada con nacimiento estelar (solo una vez) ───────────
function InsigniaAnimada({ size = 120 }) {
  const svgRef = useRef(null);

  useEffect(() => {
    const svg = svgRef.current;
    if (!svg) return;

    const STARS = [
      { id: 'an-betelgeuse', ox: 60,  oy: 60  },
      { id: 'an-bellatrix',  ox: -50, oy: 50  },
      { id: 'an-mintaka',    ox: 35,  oy: -45 },
      { id: 'an-alnilam',    ox: -40, oy: -35 },
      { id: 'an-alnitak',    ox: -35, oy: 40  },
      { id: 'an-saiph',      ox: 55,  oy: -50 },
      { id: 'an-rigel',      ox: -55, oy: -40 },
    ];

    const LINES = ['an-l1','an-l2','an-l3','an-l4','an-l5','an-l6','an-l7','an-l8'];
    const FLASHES = ['an-fl1','an-fl2','an-fl3','an-fl4','an-fl5','an-fl6','an-fl7'];
    const HALOS = ['an-hb','an-hr'];
    const DESTS = ['an-db','an-dr'];

    const STAR_DUR = 900;
    const STAR_DELAY = 200;
    const LINE_START = STARS.length * STAR_DELAY + STAR_DUR + 200;
    const LINE_DUR = 380;
    const LINE_DELAY = 120;

    function ease(t) { return t < 0.5 ? 4*t*t*t : 1 - Math.pow(-2*t+2,3)/2; }
    function easeOut(t) { return 1 - Math.pow(1-t,3); }

    function animate({ duration, delay = 0, onProgress, onDone }) {
      const start = performance.now() + delay;
      function frame(now) {
        const elapsed = now - start;
        if (elapsed < 0) { requestAnimationFrame(frame); return; }
        const t = Math.min(1, elapsed / duration);
        onProgress(t);
        if (t < 1) requestAnimationFrame(frame);
        else if (onDone) onDone();
      }
      requestAnimationFrame(frame);
    }

    function fadeIn(el, dur, delay) {
      animate({ duration: dur, delay, onProgress(t) { el.setAttribute('opacity', easeOut(t)); } });
    }

    function animateFlash(el) {
      animate({
        duration: 550,
        onProgress(t) {
          const r = easeOut(t) * 26;
          const op = t < 0.15 ? t/0.15 : 1-(t-0.15)/0.85;
          el.setAttribute('r', r);
          el.setAttribute('opacity', op * 0.9);
        },
        onDone() { el.setAttribute('opacity', 0); }
      });
    }

    // Animar estrellas
    STARS.forEach((s, i) => {
      const el = svg.getElementById(s.id);
      if (!el) return;
      const flashEl = svg.getElementById(FLASHES[i]);

      const finalX = parseFloat(el.getAttribute('data-fx') || 0);
      const finalY = parseFloat(el.getAttribute('data-fy') || 0);
      const cx = (finalX + s.ox + finalX) / 2 + (finalY - s.oy) * 0.3;
      const cy = (finalY + s.oy + finalY) / 2 + (s.ox - finalX) * 0.3;

      animate({
        duration: STAR_DUR,
        delay: i * STAR_DELAY,
        onProgress(t) {
          const et = ease(t);
          const bx = (1-et)*(1-et)*s.ox + 2*(1-et)*et*(cx-finalX) + et*et*0;
          const by = (1-et)*(1-et)*s.oy + 2*(1-et)*et*(cy-finalY) + et*et*0;
          const sc = 0.05 + 0.95*et;
          el.setAttribute('opacity', Math.min(1, t * 3));
          el.setAttribute('transform', `translate(${bx},${by}) scale(${sc})`);
          if (el.style) el.style.transformOrigin = `${finalX}px ${finalY}px`;
        },
        onDone() {
          el.removeAttribute('transform');
          el.setAttribute('opacity', 1);
          if (flashEl) animateFlash(flashEl);
          if (i === 0) { setTimeout(() => fadeIn(svg.getElementById('an-hb'), 600, 0), 100); setTimeout(() => fadeIn(svg.getElementById('an-db'), 300, 200), 100); }
          if (i === 6) { setTimeout(() => fadeIn(svg.getElementById('an-hr'), 600, 0), 100); setTimeout(() => fadeIn(svg.getElementById('an-dr'), 300, 200), 100); }
        }
      });
    });

    // Animar líneas
    LINES.forEach((id, li) => {
      const el = svg.getElementById(id);
      if (!el) return;
      const isDash = li === 7;
      const delay = LINE_START + li * LINE_DELAY;

      if (isDash) {
        animate({ duration: LINE_DUR, delay, onProgress(t) { el.setAttribute('opacity', easeOut(t)); } });
      } else {
        const len = Math.hypot(
          el.x2.baseVal.value - el.x1.baseVal.value,
          el.y2.baseVal.value - el.y1.baseVal.value
        );
        el.setAttribute('opacity', 1);
        el.style.strokeDasharray = len;
        el.style.strokeDashoffset = len;
        animate({
          duration: LINE_DUR, delay,
          onProgress(t) { el.style.strokeDashoffset = len * (1 - easeOut(t)); }
        });
      }
    });

  }, []);

  return (
    <svg
      ref={svgRef}
      viewBox="-24 -24 173 173"
      width={size}
      height={size}
      style={{ display: 'block' }}
    >
      <defs>
        <radialGradient id="an-gb" cx="35%" cy="35%" r="65%">
          <stop offset="0%" stopColor="#FF2DA6"/>
          <stop offset="100%" stopColor="#A855F7"/>
        </radialGradient>
        <radialGradient id="an-gr" cx="35%" cy="50%" r="65%">
          <stop offset="0%" stopColor="#FFFFFF"/>
          <stop offset="50%" stopColor="#00C8FF"/>
          <stop offset="100%" stopColor="#A855F7"/>
        </radialGradient>
        <radialGradient id="an-hb" cx="50%" cy="50%" r="50%">
          <stop offset="0%" stopColor="#FF2DA6" stopOpacity="0.35"/>
          <stop offset="100%" stopColor="#FF2DA6" stopOpacity="0"/>
        </radialGradient>
        <radialGradient id="an-hr-g" cx="50%" cy="50%" r="50%">
          <stop offset="0%" stopColor="#00C8FF" stopOpacity="0.4"/>
          <stop offset="100%" stopColor="#00C8FF" stopOpacity="0"/>
        </radialGradient>
        <radialGradient id="an-flash" cx="50%" cy="50%" r="50%">
          <stop offset="0%"   stopColor="#FFFFFF" stopOpacity="1"/>
          <stop offset="100%" stopColor="#A855F7" stopOpacity="0"/>
        </radialGradient>
        <filter id="an-glow">
          <feGaussianBlur stdDeviation="2.5" result="b"/>
          <feMerge><feMergeNode in="b"/><feMergeNode in="SourceGraphic"/></feMerge>
        </filter>
        <filter id="an-glow-sm">
          <feGaussianBlur stdDeviation="1" result="b"/>
          <feMerge><feMergeNode in="b"/><feMergeNode in="SourceGraphic"/></feMerge>
        </filter>
      </defs>

      {/* Halos */}
      <circle id="an-hb" cx="24"  cy="14"  r="22" fill="url(#an-hb)"  opacity="0"/>
      <circle id="an-hr" cx="100" cy="100" r="22" fill="url(#an-hr-g)" opacity="0"/>

      {/* Líneas */}
      <line id="an-l1" x1="24"  y1="14"  x2="88"  y2="26"  stroke="#A855F7" strokeWidth="1.2" strokeLinecap="round" opacity="0"/>
      <line id="an-l2" x1="24"  y1="14"  x2="34"  y2="60"  stroke="#A855F7" strokeWidth="1.2" strokeLinecap="round" opacity="0"/>
      <line id="an-l3" x1="88"  y1="26"  x2="78"  y2="52"  stroke="#A855F7" strokeWidth="1.2" strokeLinecap="round" opacity="0"/>
      <line id="an-l4" x1="34"  y1="60"  x2="56"  y2="58"  stroke="#00C8FF" strokeWidth="2"   strokeLinecap="round" opacity="0" filter="url(#an-glow-sm)"/>
      <line id="an-l5" x1="56"  y1="58"  x2="78"  y2="52"  stroke="#00C8FF" strokeWidth="2"   strokeLinecap="round" opacity="0" filter="url(#an-glow-sm)"/>
      <line id="an-l6" x1="34"  y1="60"  x2="20"  y2="108" stroke="#A855F7" strokeWidth="1.2" strokeLinecap="round" opacity="0"/>
      <line id="an-l7" x1="78"  y1="52"  x2="100" y2="100" stroke="#A855F7" strokeWidth="1.2" strokeLinecap="round" opacity="0"/>
      <line id="an-l8" x1="20"  y1="108" x2="100" y2="100" stroke="#00C8FF" strokeWidth="0.8"
        strokeDasharray="3,4" strokeLinecap="round" opacity="0"/>

      {/* Estrellas */}
      <g id="an-betelgeuse" opacity="0" data-fx="24" data-fy="14" filter="url(#an-glow)">
        <circle cx="24" cy="14" r="10" fill="url(#an-gb)"/>
      </g>
      <g id="an-bellatrix" opacity="0" data-fx="88" data-fy="26">
        <circle cx="88" cy="26" r="7" fill="#A855F7"/>
      </g>
      <g id="an-mintaka" opacity="0" data-fx="34" data-fy="60">
        <circle cx="34" cy="60" r="5" fill="#00C8FF"/>
      </g>
      <g id="an-alnilam" opacity="0" data-fx="56" data-fy="58">
        <circle cx="56" cy="58" r="5" fill="#00C8FF"/>
      </g>
      <g id="an-alnitak" opacity="0" data-fx="78" data-fy="52">
        <circle cx="78" cy="52" r="5" fill="#00C8FF"/>
      </g>
      <g id="an-saiph" opacity="0" data-fx="20" data-fy="108">
        <circle cx="20" cy="108" r="6" fill="#A855F7"/>
      </g>
      <g id="an-rigel" opacity="0" data-fx="100" data-fy="100" filter="url(#an-glow)">
        <circle cx="100" cy="100" r="10" fill="url(#an-gr)"/>
        <circle cx="100" cy="100" r="10" fill="none" stroke="#0A0A12" strokeWidth="0.8"/>
        <text x="100" y="104" textAnchor="middle"
          fontFamily="Georgia,serif" fontSize="11" fontWeight="700" fill="#0A0A12">S</text>
      </g>

      {/* Flashes */}
      <circle id="an-fl1" cx="24"  cy="14"  r="1" fill="url(#an-flash)" opacity="0"/>
      <circle id="an-fl2" cx="88"  cy="26"  r="1" fill="url(#an-flash)" opacity="0"/>
      <circle id="an-fl3" cx="34"  cy="60"  r="1" fill="url(#an-flash)" opacity="0"/>
      <circle id="an-fl4" cx="56"  cy="58"  r="1" fill="url(#an-flash)" opacity="0"/>
      <circle id="an-fl5" cx="78"  cy="52"  r="1" fill="url(#an-flash)" opacity="0"/>
      <circle id="an-fl6" cx="20"  cy="108" r="1" fill="url(#an-flash)" opacity="0"/>
      <circle id="an-fl7" cx="100" cy="100" r="1" fill="url(#an-flash)" opacity="0"/>

      {/* Destellos */}
      <g id="an-db" opacity="0">
        <line x1="24" y1="1"  x2="24" y2="5"  stroke="#FF2DA6" strokeWidth="0.8" strokeLinecap="round"/>
        <line x1="24" y1="23" x2="24" y2="27" stroke="#FF2DA6" strokeWidth="0.8" strokeLinecap="round"/>
        <line x1="11" y1="14" x2="15" y2="14" stroke="#FF2DA6" strokeWidth="0.8" strokeLinecap="round"/>
        <line x1="33" y1="14" x2="37" y2="14" stroke="#FF2DA6" strokeWidth="0.8" strokeLinecap="round"/>
      </g>
      <g id="an-dr" opacity="0">
        <line x1="100" y1="87"  x2="100" y2="91"  stroke="#00C8FF" strokeWidth="0.8" strokeLinecap="round"/>
        <line x1="100" y1="109" x2="100" y2="113" stroke="#00C8FF" strokeWidth="0.8" strokeLinecap="round"/>
        <line x1="87"  y1="100" x2="91"  y2="100" stroke="#00C8FF" strokeWidth="0.8" strokeLinecap="round"/>
        <line x1="109" y1="100" x2="113" y2="100" stroke="#00C8FF" strokeWidth="0.8" strokeLinecap="round"/>
      </g>
    </svg>
  );
}
